@dlt.source
def odoo_source():

    db, pwd, usr = secrets.get('odooDB'), secrets.get('odooPassword'), secrets.get('odooUser')
    uid = OdooAPI.get_uid(url, db, usr, pwd)

    logger.info(f'Pipeline resource generations for', extra={'tables': str(tables), 'priimary_keys': str(pks)})
    def create_resource(table: str, pk: str, schema: str = None):
        skma = schema+'_' if schema else ''
        updated = dlt.sources.incremental("write_date", initial_value="1970-01-01 00:00:00")

        @dlt.resource(name=f"{skma}{table.replace('.','_')}", write_disposition="merge", primary_key=[pk, "_source"])
        def resource(updated=updated):
            domain = [("write_date", ">", updated.last_value)] if updated.last_value else []
            logger.info(f'Processing table {table} start', extra={'stage': 'data-fetching', 'connection_name': connection_name})
            for page in OdooAPI.paginated_search_read(url, db, uid, pwd, table, domain):
                for row in page:
                    [row["_source"], row["_e2e_pk"], row["_e2e_src_db"]] = [url, f'{row["id"]}_{url}', url]

                yield page
            logger.info(f'Processing table {table} end', extra={'stage': 'data-fetching-completed', 'connection_name': connection_name})
        return resource()

    return [create_resource(tables[x].replace('/',''), pks[x], 'public') for x in range(len(tables))]
